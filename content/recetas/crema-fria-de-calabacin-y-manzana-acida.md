categories = [
  "Sopas y cremas",
]
author = "Amama"
difficulty = "Baja"
time = "30-60 min"
servings = "4 personas"
ingredients = [
  "3 calabacines grandes",
  "400ml de caldo de verduras",
  "2 manzanas granny smith",
  "2 dientes de ajo",
  "3 cebollas dulces",
  "125gr de yogur natural de cabra",
  "1 cucharada grande orégano",
  "150gr de butifarra o salchicha frescas de carnicería (opcional)",
  "Aceite de oliva virgen extra",
  "Sal",
  "Pimienta",
]
steps = [
  "Calentar una olla con aceite de oliva virgen extra.",
  "Sofreír la cebolla cortada a fuego bajo, con la olla tapada e ir agregando agua de vez en cuando, durante media hora.",
  "Pelar, despepitar y cortar las manzanas.",
  "Cortar el calabacín en cubos y picar ajos.",
  "Añadir todo a la olla junto con el caldo de verduras y cocinar suavemente durante 45'",
  "Apagar el fuego y esperar hasta que atempere.",
  "Batir junto con el orégano y el yogur.",
  "Corregir el punto de sal.",
  "Guardar en la nevera 2 horas.",
  "Sacar 10' antes de servir y acompañar con un poco de butifarra o salchicha doradas en la sartén.",
]
+++

Se puede poner cualquier yogur si no hay de cabra.
