difficulty = "Baja"
time = "30-60 min"
servings = "4 personas"
ingredients = [
  "4 lomos de bacalao desalado",
  "1 coliflor pequeña",
  "2 patatas medianas",
  "4 dientes de ajo",
  "1 cebollas grande blanca o dulce",
  "2 cucharaditas de pimentón dulce o agridulce",
  "10ml de manzana o sidra (o al gusto)",
  "Sal",
  "Aceite de oliva virgen extra",
  "Perejil fresco picado",
]
steps = [
  "Limpiar y cortar la coliflor en floretes.",
  "Pelar y cortar las patatas en rodajas del grosor de un dedo.",
  "Cortar la cebolla en juliana fina. y laminar los ajos.",
  "Calentar una sartén con aceite de oliva, dorar los ajos ligeramente, añadir la cebolla y una pizca de sal. Remover y sofreír hasta que estetiernos.",
  "Retirar del fuego, añadir el vinagre y el pimentón, remover y reservar.",
  "En una cazuela amplia calentar agua, añadir las patatas y la coliflor con un poco de sal.",
  "Cocer  hasta que estén tiernos, pero firmes y que no se deshagan.",
  "Añadir el bacalao por  encima, tapar de nuevo y dejar cocinar durante 5'-7'.",
  "Escurrir el agua.",
  "Añadir la salsa de pimentón por encima, remover, dejar reposar unos minutos y servir con perejil al gusto.",
]
