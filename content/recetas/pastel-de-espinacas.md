ingredients = [
  "400 gr de espinacas.",
  "Champiñones.",
  "2 puerros.",
  "2 dientes de ajo.",
  "Sal en escamas con boletus ( o sal normal).",
  "Pimienta negra.",
  "3 huevos.",
  "400 gr de queso ricotta.",
  "10-12 hojas de masa filo.",
  "25 gr de piñones.",
  "Aceite de oliva Virgen extra.",
  "Mantequilla.",
]
steps = [
  "Cocer las espinacas con agua y sal.",
  "En una cazuela poner aceite, incorporar los ajos y puerros troceados.",
  "Sofreír a fuego medio-bajo hasta ablandar.",
  "Añadir al sofrito las espinacas y mezclar.",
  "Retirar del fuego y añadir los huevos, el ricotta, sal, pimienta, los champiñones y piñones.",
  "Mezclar todo.",
  "Precalentar el horno a 200º",
  "Engrasar un molde con mantequilla, colocar una hoja de masa filo, pintar con mantequilla y seguir colocando masa filo de la misma manera hasta formar la costra del pastel.",
  "Verter el relleno e ir cerrando con el sobrante de filo hasta tener cubierto todo el relleno.",
  "Pintar con mantequilla y hornear hasta dorar.",
  "Sacar del horno, dejar enfriar y desmoldar.",
]
