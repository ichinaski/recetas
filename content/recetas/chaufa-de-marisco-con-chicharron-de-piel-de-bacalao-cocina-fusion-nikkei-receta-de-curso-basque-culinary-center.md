ingredients = [
  "Para el arroz de chaufa:",
  "300gr de arroz jazmín cocido frío",
  "120gr de gamba pelada",
  "120 gr de pulpo cocido",
  "80gr de cebolleta picada",
  "40gr de jengibre picado",
  "15gr de ajo picado",
  "2 huevos",
  "30ml de salsa de soja",
  "40ml de salsa teriyaki",
  "15ml de aceite de sésamo",
  "20ml de aceite vegetal",
  "10gr de cilantro picado",
  "Sal",
  "Pimienta",
  "Para el chicharrón de piel de bacalao:",
  "200gr de piel de bacalao limpia",
  "Aceite para freir",
  "Sal",
  "Acabado y presentación:",
  "10ml  de aceite de sésamo",
  "20gr de cebollino picado",
  "1 lima en gajos",
  "Brotes frescos",
  "Sésamo tostado",
]
steps = [
  "Calentar el wok con le aceite vegetal.",
  "Saltear las gambas a fuego fuerte hasta dorar ligeramente.",
  "Incorporar el pulpo cortado en rodajas y saltar brevemente.",
  "Añadir el ajo, el jengibre y la cebolleta.",
  "Cocinar unos segundos sin dejar colorear en exceso.",
  "Incorporar el huevo y remover rápidamente.",
  "Añadir el arroz frío y saltear a fuego vivo separando bien los granos.",
  "Agregar la salsa de soja, la salsa teriyaki y el aceite de sésamo.",
  "Saltear hasta caramelizar ligeramente el arroz.",
  "Rectificar sal y pimienta.",
  "Terminar con cilantro picado.",
  "Para el chicharrón de pile de bacalao, limpiar las pieles del bacalao retirando restos de espinas y exceso de grasa.",
  "Secar completamente con papel absorbente.",
  "Cortar en piezas regulares.",
  "Freír en aceite caliente a 180º hasta obtener una textura inflada y crujiente.",
  "Escurrir sobre papel absorbente.",
  "Reservar en seco.",
  "Acabado y presentación:",
  "Disponer el arroz chaufa en un bol o plato hondo.",
  "Colocar encima el chicharrón de piel de bacalao para mantener el crujiente.",
  "Añadir unas gotas de aceite de sésamo.",
  "Terminar con cebollino, sésamo tostado y brotes frescos.",
  "Servir con lima aparte.",
  "Se sirve caliente.",
]

Utensilios: wok, sarten, espumadera, papel absorbente, cuchillo, tabla de corte y bol.
