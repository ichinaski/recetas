title = "Buñuelos de bacalao"
date = "2024-09-17T20:28:49"
categories = [
  "Pescado",
]
author = "Amama"
difficulty = "Baja"
time = "30-60 min"
ingredients = [
  "300 gr de bacalao desalado",
  "150 gr de mantequilla",
  "300 ml de agua",
  "1/2 cucharadita de sal",
  "210 gr de harina de trigo",
  "5 gr de levadura química",
  "3 huevos",
  "2 dientes de ajo",
  "Aceite de oliva virgen extra",
  "Cebollino fresco (opcional)",
]
steps = [
  "Preparación de la masa:",
  "Calentar agua en un cazo y dar un hervor a las migas de bacalao (2 minutos).",
  "Escurrir y secar,  dejar que se atempere antes de desmenuzarlo. Reservar.",
  "En una cacerola calentar los 300 ml de agua y la mantequilla.",
  "Cuando empiece a hervir, retirar del fuego y echar la harina de trigo y la levadura tamizados, con la levadura de un solo golpe.",
  "Llevar de nuevo al fuego y remover con cuchara de madera hasta conseguir un masa homogénea y sin grumos.",
  "Retirar del fuego y dejar que la masa se atempere antes de añadir los huevos, de uno en uno, removiendo bien hasta integrar antes de incorporar el siguiente.",
  "Tiene que quedar una masa espesa, cremosa y brillante.",
  "Rallar los dientes de ajo y añadir a la masa junto con el bacalao.",
  "Remover y dejar enfriar en la nevera 1 hora como mínimo.",
  "Freír los buñuelos:",
  "Calentar abundante aceite en una sartén o cestillo para freír y dejar caer pequeñas cantidades  de masa en su interior. Voltear para que se  hagan bien, pero cremosos.",
  "Escurrir los buñuelos y depositar en un plato con papel absorbente.",
  "Espolvorear con cebollino fresco picado ( opcional).",
]
+++

Se pueden hacer en freidora de aire y tendrán menos calorías.
