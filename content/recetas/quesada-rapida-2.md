title = "Quesada rápida"
date = "2026-02-22T19:00:37"
categories = [
  "Postres",
]
author = "Amama"
difficulty = "Baja"
time = "30-60 min"
ingredients = '''
300gr de requeson o de ricota
3 huevos
2 vasos de agua de leche entera (unos 400-500ml)
2 vasos de agua de azúcar 
1 vaso de agua de harina
Mantequilla'''
preparation = '''
Precalentar el horno a 200º
Engrasar un  molde con mantequilla. 
Batir todos los ingredientes con la batidora hasta que quede una masa homogénea.
Verter en el molde y hornear unos 50' o hasta que esté hecha en el centro.
Servir fría.'''
