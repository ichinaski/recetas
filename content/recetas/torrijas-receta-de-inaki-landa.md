date = "2026-04-24T21:02:38"
categories = [
  "Postres",
]
author = "Amama"
difficulty = "Baja"
time = "30-60 min"
ingredients = [
  "2 barras de pan (reposadas del día anterior), lo más claras y lisas posibles",
  "2 litros de leche entera",
  "2 ramas de canela",
  "La corteza de 1 limón o naranja",
  "2 copas de anís",
  "2 cucharas de vainilla",
  "340gr de azúcar",
  "Canela en polvo",
  "Huevo para rebozar.",
]
steps = [
  "Cocer la leche con las ramas de canela, la vainilla, la corteza de limón (o naranja) y el anís.",
  "Trocear el pan en rebanadas de 22/25 mm y colocar en una bandeja.",
  "Añadir el azúcar a la leche al final de la cocción, para evitar que se pegue en la cazuela.",
  "Dejar templar  la cocción de leche y echarla por encima de las rebanadas de pan.",
  "Dejar que se empapen bien, unos 30'. Pasar por huevo y freir en aceite, con cuidado de que no se quemen.",
  "Cuando estén frías,  si se quiere se pueden espolvorear con canela molida.",
]
