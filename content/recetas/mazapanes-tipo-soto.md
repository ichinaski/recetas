author = "Amama"
difficulty = "Baja"
time = "30-60 min"
servings = "Indeterminado"
ingredients = '''
1 Kg de almendra molida
1 kg de azúcar glass 
1/2 kg de patata cocida en puré
1 cucharada (café) de ácido de limón o la ralladura de medio limón'''
preparation = '''
Mezclar todos los ingredientes
Hacer bolas del mismo tamaño
Pintar con almíbar
Introducir en el horno al grill y esperar hasta que se doren
Para hacer el almíbar: 1/2 vaso de agua + 1 vaso de azúcar. 
Poner al fuego durante unos minutos hasta que vaya haciéndose más compacta
Cuando se hayan enfriado, envolver en papel de polvorón.'''
