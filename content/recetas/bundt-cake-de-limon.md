difficulty = "Media"
time = "30-60 min"
ingredients = [
  "350gr de harina",
  "225gr de mantequilla a temperatura ambiente",
  "300gr de azúcar",
  "250ml de leche",
  "2 limones",
  "4 huevos",
  "2 cucharaditas de levadura",
  "Una pizca de sal",
  "120gr de azúcar glas",
]
steps = [
  "Precalentar el horno a 180º",
  "Engrasar el molde rellenando bien,  se puede utilizar un pincel.",
  "Rallar la piel de los limones y añadir el zumo de uno de ellos, remover y dejar reposar.",
  "Batir la mantequilla con el azúcar durante 5' para que se airee la mezcla y quede esponjoso el bizcocho.",
  "Sin dejar de batir, añadir uno a uno los huevos, que estén a temperatura ambiente y no incorporar el siguiente hasta que esté bien integrado el anterior.",
  "Tamizar la harina con la levadura y la sal.",
  "Añadir 1/3 de la harina a la mezcla anterior con movimientos envolventes y suaves.",
  "Añadir la mitad de la leche, moviendo de la misma manera y así hasta terminar con los ingredientes, empezando y terminado con la harina.",
  "Incorporar la ralladura reservada y volcar la mezcla en el molde.",
  "Golpear un poco el molde sobre la encimera para ayudar a salir las burbujas de la mezcla.",
  "Hornear a 170º alrededor de 1 hora o hasta que pinchemos con una aguja y salga limpia.",
  "Sacar del horno y dejar entibiar en una rejilla 15'",
  "Desmoldar y dejar enfriar.",
  "Para preparar el glaseado, añadir el zumo de limón al azúcar glas poco a poco, removiendo continuamente. Si vemos que no es suficiente, añadir una cucharadita de agua.",
  "Volcar la glasa sobre el bizcocho, repartiéndolo uniformemente.",
]
