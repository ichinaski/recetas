categories = [
  "Pescado",
]
author = "Amama"
difficulty = "Baja"
time = "30-60 min"
servings = "4 personas"
ingredients = [
  "400 gr de bacalao desalado.",
  "1 patata mediana.",
  "3 dientes de ajo.",
  "150-200 ml de leche entera.",
  "Pimentón De la Vera ( opcional ).",
  "Perejil,fresco picado ( opcional ).",
  "Aceite de oliva.",
  "Sal.",
]
steps = [
  "Escurrir y secar bien el bacalao.",
  "Cocer la patata y pelarla.",
  "Colocar aceite en un cazo y poner los ajos fileteados hasta que se doren.",
  "Bajar el fuego al mínimo colocar el bacalao, dejar que se cocine lentamente tapado 2’ por cada lado o hasta que se dore.",
  "Sacar el bacalao del aceite y majarlo bien en un mortero junto con los ajos y la patata cocida troceada.",
  "Añadir 1 cucharada de aceite de la fritura e ir añadiendo leche tibia hasta integrarla.",
  "Seguir añadiendo otra cucharada de aceite y más leche hasta conseguir una mezcla homogénea.",
  "Rectificar el punto de sal.",
  "Servir espolvoreado con perejil o pimentón De la Vera y aderezado con un poco de aceite de freír los ajos.",
  "Acompañar con toscas, frutos secos y aceitunas negras.",
]
