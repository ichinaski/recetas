author = "Amama"
difficulty = "Baja"
time = "Menos de 30 min"
servings = "2 personas"
ingredients = [
  "2 naranjas",
  "2 limones",
  "2 huevos",
  "1 trozo de chorizo de unos 60gr",
  "80gr de jamón ibérico en lonchas finas",
  "un chorrito de vinagre de vino",
  "Aceite de oliva Virgen extra",
  "Sal",
]
steps = [
  "Cocer los huevos 7' y pelar.",
  "Pelar el chorizo y cortarlo en daditos, si se quiere se puede pasar por la sartén o el microondas para que sude un poco.",
  "Pelar los limones y las naranjas, sacar los gajos y disponerlos en un plato.",
  "Preparar una vinagreta con el vinagre, aceite de oliva al gusto y un poco de sal (El jamón y el chorizo ya le aportarán la suya).",
  "Poner los huevos troceados y el chorizo sobre los cítricos.",
  "Aliñar con la vinagreta y rematar con el jamón.",
]

EL huevo puede ir frito en lugar de cocido.
El chorizo puede ir cocido, frito o crudo.
