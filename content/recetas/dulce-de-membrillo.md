  "Postres",
]

difficulty = "Baja"
time = "30-60 min"
ingredients = [
  "1kg de membrillo",
  "800 gr de azúcar",
  "1/2 vaso de agua",
  "Zumo de 1/2 limón",
  "1 rama de canela",
]
steps = [
  "Pelar y trocear los membrillos, ponerlos a cocer a  fuego lento con 1/2 vaso de agua, hasta que cojan color, en ese momento agregar el azúcar,  el zumo de limón y la rama de canela.",
  "Dejar a fuego medio hasta que se deshaga el azúcar y coja el color deseado.",
  "Batir con la batidora y poner en el recipiente donde se vaya a guardar.",
]
