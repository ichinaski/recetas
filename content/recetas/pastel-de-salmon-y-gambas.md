title = "Pastel de salmón y gambas"
date = "2024-05-18T16:21:10"
categories = [
  "Pescado",
]
author = "Amama"
difficulty = "Baja"
time = "60-120 min"
servings = "8 personas"
ingredients = [
  "Un brick  pequeño de leche evaporada.",
  "250 gr de salmón fresco limpio de piel y espinas.",
  "400 gr de colas peladas.",
  "150 gr de puerro ( la parte blanca).",
  "150 gr de cebolla tierna.",
  "100 gr de calabaza.",
  "200 ml de caldo de pescado.",
  "15 gr de alcaparras.",
  "8 huevos.",
  "3 cucharadas de aceite de oliva.",
  "Pimienta negra molida.",
  "Ralladura de 1/2 limón.",
]
steps = [
  "Picar el puerro y la cebolleta y cortar la calabaza en daditos pequeños.",
  "Cortar los lomos de salmón en trozos de 2  cm aproximadamente y las gambas, reservando sin cortar 50 gr de la superficie gambas más grandes.",
  "Marcar el puerro y la cebolleta con 2 cucharadas de aceite de oliva y una pizca de sal a fuego medio hasta que estén transparentes, unos 10’.",
  "Marcar los dados de calabaza con 1 cucharada de aceite durante 5’.",
  "Añadir al sofrito de puerro y cebolleta el salmón y las gambas y cocinar a fuego medio durante 5’.",
  "Cuando el salmón y las gambas estén cocinadas, reservar las gambas de decoración y añadir el caldo de pescado y cocinar 10’-15’ más hasta que se evapore casi todo el líquido.",
  "Apagar el fuego y añadir los daditos de calabaza, la ralladura de limón y la pimienta.",
  "Probar y rectificar de sal y pimienta.",
  "Dejar templar.",
  "Precalentar el horno a 180º.",
  "Batir los huevos con la leche evaporada, añadir el sofrito de pescado y mezclar.",
  "Verter la mezcla en un molde de 30 cm. engrasado y forrado con papel de horno.",
  "Hornear durante unos 45’.",
  "Dejar reposar unos minutos en el molde y desmoldar.",
  "Decorar con las gambas y las alcaparras.",
]
