categories = [
  "Aperitivos y tapas",
]
author = "Amama"
difficulty = "Baja"
time = "30-60 min"
ingredients = [
  "80 gr de mantequilla.",
  "120 gr de harina.",
  "1 litro de leche.",
  "1 ajo.",
  "150 gr de jamón.",
  "Huevos.",
  "Pan rallado.",
]
steps = [
  "En una sartén poner la mantequilla con el ajo.",
  "Cuando esté dorado y a fuego suave ir añadiendo al harina poco a poco y remover. Dejar que la harina se haga unos minutos.",
  "Añadir la leche templada poco a poco removiendo constantemente.",
  "Cuando se haya hecho, añadir el jamón y remover unos minutos. Si se  quiere hacer de otros ingredientes es el mismo procedimiento.",
  "Probar el punto de sal.",
  "Pasar el contenido a una fuente y tapar con papel film en contacto con la superficie, para que no se haga costra.",
  "Dejar enfriar guardar en el frigorífico hasta el día siguiente.",
  "Hacer las croquetas ayudándose de harina, pasar por huevo batido y después pan rallado.",
]
