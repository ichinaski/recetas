difficulty = "Media"
time = "30-60 min"
servings = "20 pasteles"
ingredients = [
  "Crema:",
  "500ml de leche",
  "500ml de almibar",
  "80gr de yema de huevo",
  "1 huevo entero",
  "50gr de harina",
  "25r de almidón de maíz (maicena)",
  "Una placa de hojaldre envasada o casera",
  "Almibar:",
  "250ml de agua",
  "400gr de azúcar",
  "Piel de 1 limón",
  "1 palo pequeño de canela",
]
steps = [
  "Para el almíbar:",
  "Hervir el agua con la piel de limón y el palo de canela.",
  "Añadir el azúcar,  cocer hasta que el azúcar desaparezca y dejar enfriar.",
  "Cortar la masa de hojaldre en 4 rectángulos de 40x25cm y enrollar cada uno para obtener así 4 rulos de 40cm.",
  "Guardar en la nevera como mínimo 4 horas.",
  "Sacar la masa de la nevera y de cada rollo cortar 25 trozos de cerca de 18/20gr. Salen 25 pastelitos de cada rollo.",
  "Poner los trocitos de la masa dentro de los moldes, y con los dedos mojados en agua abrir la masa en el molde.",
  "Repetimos este proceso con todos los pastelitos.",
  "Dejar la masa descansar dentro de los moldes como mínimo 1 hora.",
  "Para la crema, hervir 375 ml de leche y mezclar los restantes 125ml de leche con la harina y la maicena.",
  "Añadir la leche hirviendo a la mezcla anterior.",
  "Añasir 500ml de almibar en 3 partes mezclando bien entre cada adición.",
  "Añadir las yemas y los huevos mezclando bien.",
  "Poner la crema dentro de los moldes, hornear a 250º unos 5' en medio del horno para que se cocine bien el hojaldre.",
  "A media cocción dar la vuelta a la bandeja y la subimos para que esté más cerca del grill y tueste por arriba los pasteles, unos 5' más.",
]
