categories = [
  "Verduras",
]
author = "Amama"
difficulty = "Baja"
time = "30-60 min"
servings = "2 personas"
ingredients = [
  "350gr de espárragos blancos frescos",
  "30gr de mantequilla",
  "1 cucharadita de azúcar",
  "Sal y pimienta recién molida",
  "2 huevos",
]
steps = [
  "Limpiar y pelar los espárragos.",
  "Filetear con un cuchillo afilado en sentido oblicuo, en rodajas de de aproximadamente 1/2 cm, dejando las yemas enteras.",
  "Derretir la mantequilla en una sartén, incorporar los espárragos con las yemas y el azúcar.",
  "Caramelizar suavemente a fuego medio, removiendo de vez en cuando, durante al menos10' o hasta que se ablanden.",
  "Cuando estén tiernos,  sazonar con sal y pimienta antes de servir.",
  "Si se quiere, acompañar con un huevo escalfado o frito por persona. Servir sobre los espárragos tibiios.",
]
