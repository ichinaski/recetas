title = "Ensalada de calabacín,  queso feta y nueces"
date = "2026-08-20T08:16:05"
categories = [
  "Ensaladas",
]
author = "Amama"
difficulty = "Baja"
time = "Menos de 30 min"
servings = "4 personas"
ingredients = [
  "2 calabacines medianos",
  "150gr de queso feta",
  "4 cucharadas de nueces peladas",
  "Unas ramitas de eneldo",
  "1 limón",
  "Aceite de oliva virgen extra",
  "Sal y pimienta recién molida",
]
steps = [
  "Cortar el calabacín en rodajas o lámina finas (1-2 mm).",
  "Ponerlos en un bol, añadir 4 cucharadas de aceite de oliva,  el zumo de 1/2 limón,  pimienta negra y un poco de sal.",
  "Remover bien y dejar marinando entre 15'-30'",
  "Picar las nueces y el eneldo.",
  "Cuando haya pasado el tiempo marinado, escurrir bien el calabacín y ponerlo en una fuente de servir.",
  "Pasar a un bote lo que haya quedado del aliño.",
  "Añadir 2 cucharadas más de aceite,  otro chorro de zumo de limón y un poco más de sal.",
  "Cerrar el bote y mover para que emulsione.",
  "Verter el aliño sobre el calabacin.",
  "Repartir por encima el queso feta desmigado con las manos, las nueces y por último el eneldo.",
]
