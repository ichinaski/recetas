title = "Pasta con calabaza asada y crema de parmesano"
date = "2025-10-29T20:00:07"
categories = [
  "Pasta",
]
author = "Amama"
difficulty = "Baja"
time = "30-60 min"
ingredients = [
  "300gr de pasta fresca",
  "500gr de calabaza cacahuete (pelada y cortada en dados)",
  "1/2 calabaza cacahuete cortada en rodajas (sin pelar)",
  "Aceite de oliva Virgen extra",
  "Tomillo fresco",
  "2 dientes de ajo",
  "500ml de nata para cocinar ( o leche evaporada)",
  "100gr de parmesano rallado",
  "Sal y pimienta negra recien molida",
  "Hojas de salvia fritas (para decorar)",
]
steps = [
  "Precalentar el horno a 200º",
  "Colocar las rodajas de calabaza en una bandeja de horno, rociar con un chorro de aceite de oliva, poner sal y pimienta negra.",
  "Asar durante 35'-40' o hasta que los bordes estén dorados y la carne tierna.",
  "Para preparar la salsa:",
  "En una cazuela poner la calabaza cortada en dados junto con la nata o leche evaporada, los dientes de ajo y el tomillo.",
  "Añadir sal y pimienta al gusto, remover, tapar y cocinar 20'",
  "Destapar la cazuela y seguir cocinando hasta que la calabaza esté tierna y la crema se ha reducida a la mitad.",
  "Cocinar la pasta en agua hirviendo.",
  "Retirar los ajos, pelarlos y volver a ponerlos en la crema.",
  "Retirar las ramitas de tomillo y triturar la salsa hasta que la calabaza y la crema se hayan integrado.",
  "Añadir el queso parmesano rallado y remover con suavidad  hasta que se funda y se incorpore a la salsa.",
  "Poner una cucharada de salsa en cada plato hondo, repartir la pasta y añadir más salsa por encima.",
  "Servir caliente con unas hojas de salvia fritas o pipas de calabaza tostadas.",
]
