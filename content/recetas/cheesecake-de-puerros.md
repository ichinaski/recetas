title = "Cheesecake de puerros"
date = "2026-02-15T19:19:25"
categories = [
  "Verduras",
]
author = "Amama"
difficulty = "Media"
time = "60-120 min"
servings = "6-8 personas"
ingredients = [
  "500gr de puerro limpio",
  "350gr de queso crema",
  "250gr de queso ricotta o requeson",
  "200gr de nata líquida (o leche evaporada)",
  "3 huevos",
  "1 cucharada sopera de harina",
  "60gr de mantequilla",
  "1 cucharada sopera de mostaza",
  "Sal",
  "Pimienta negra recien molida",
  "Para la base:",
  "250gr de galletas cracker o saladas",
  "180gr de mantequilla fundida",
]
steps = [
  "Triturar las galletas cracker hasta obtener polvo de galleta.",
  "Añadir la mantequilla fundida y mezclar.",
  "Preparar un molde desmontable redondo de unos 21 cm de diámetro colocando papel de horno en el fondo.",
  "Verter la mezcla de galletas y mantequilla en el molde y presionar hasta obtener una capa compacta.",
  "Guardar en el frigo 30'.",
  "Precalentar el horno a 180º.",
  "Limpiar los puerros y cortar en ruedas finas.",
  "Preparar una sartén con la mantequilla, calentarla y dejar pochar los puerros con sal a fuego medio durante 20'.",
  "Batir el queso crema con la ricotta, la nata (o leche evaporada) y la mostaza, hasta obtener una masa homogénea.",
  "Incorporar los huevos, mezclar, añadir la harina y mezclar de nuevo.",
  "Agregar los puerros pochados a la crema y mezclar. Añadir pimienta negra recién molida al gusto.",
  "Verter la masa en el molde e introducir en el horno.",
  "Hornear 45'-60' o hasta que el centro del cheesecake esté cuajado.",
  "Dejar templar y durante al menos 2 horas.",
]
