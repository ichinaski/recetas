time = "30-60 min"
servings = "8 personas"
ingredients = [
  "3 huevos",
  "1 yogur",
  "1 medida del yogur de aceite de oliva",
  "2 medidas de yogur de azúcar",
  "3 medidas de yogur de harina",
  "1 sobre de levadura (16 gr de polvo para hornear)",
  "1 limón",
  "1 cucharada de azúcar glass",
  "Harina y mantequilla para untar el molde",
]
steps = [
  "Poner en un bol los huevos, el azúcar y batir con varilla manual.",
  "Añadir el yogur y el aceite.",
  "Rallar la piel de limón y echarla a la mezcla.",
  "Tamizar la harina sobre la masa y añadir la levadura química o polvo de hornear.",
  "Mezclar todo bien hasta conseguir una masa homogénea y sin grumos.",
  "Untar un molde apto para horno con un poco de mantequilla y  espolvorear con harina.",
  "Verter la masa en el molde y hornear a 180 grados durante 40´.",
  "Cuando esté hecho sacar del horno y dejar que se temple.",
  "Desmoldar y espolvorear con azúcar glass.",
  "NOTA:",
  "Para conseguir que el bizcocho suba en el horno y quede esponjoso, es importante que los ingredientes estén a temperatura ambiente y tamizar la harina.",
  "Es aconsejable utilizar la batidora de varillas manual y dejar reposar la masa 30´ antes de hornear.",
  "Es una receta básica de bizcocho, pero se pueden añadir otros ingredientes: frutos secos, chocolate negro, cacao, coco rallado, arándanos, orejones, pasas......",
