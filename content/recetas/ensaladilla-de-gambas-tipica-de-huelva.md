title = "Ensaladilla de gambas (típica de Huelva)"
date = "2024-06-09T11:49:03"
categories = [
  "Aperitivos y tapas",
]
author = "Amama"
difficulty = "Baja"
time = "30-60 min"
ingredients = [
  "400 gr de gambas.",
  "500 gr de patatas.",
  "3 huevos ( opcional).",
  "Mayonesa (de aceite de girasol) al gusto.",
  "4-5 gotas de aceite de oliva.",
  "Sal al gusto.",
  "Opcional: perejil o alguna hierba  aromática.",
]
steps = [
  "Poner una cazuela con agua y sal a hervir, cuando rompa el hervor, poner las gambas ay dejar como máximo 45 segundos.",
  "Retirar las gambas y guardar el agua.",
  "Pasar las gambas a un bol de agua con hielo y dejarlas 2’. Después de refrescarlas, escurrirlas y pelarlas.",
  "Cortar las patatas en dados y hervirlas en el agua de cocción de las gambas para potenciar el sabor. Reservar.",
  "Cocer los huevos unos 10’ y pelar.",
  "Con un tenedor  aplastar las patatas, añadir los huevos troceados y las gambas, reservando algunas gambas y trozos de huevo para decorar.",
  "Agregar 2-3 cucharadas de mayonesa y remover.",
  "Tiene que quedar jugosa, si no se puede añadir más mayonesa.",
  "Pasar la ensaladilla a una fuente y coronarla con una cucharada de mayonesa.",
  "Decorar con las gambas y trozos de huevo y rociar con unas gotas de aceite de oliva, también se puede añadir alguna hoja de perejil u otra hierba.",
]
