title = "Lasaña de calabaza, gorgonzola y frutos secos"
date = "2024-07-10T19:00:02"
categories = [
  "Pasta",
]
author = "Amama"
difficulty = "Media"
time = "30-60 min"
servings = "4-6 personas"
ingredients = [
  "Un paquete de placas para lasaña (20 placas)",
  "1,2 kg de calabaza violín sin piel ni semillas",
  "1 puerro",
  "1 cebolla",
  "75 gr de avellanas tostadas",
  "80 gr de queso parmesano o pecorino",
  "Una rama de tomillo",
  "800 ml de leche entera",
  "40 gr de harina",
  "40 gr de mantequilla",
  "Aceite de oliva virgen extra",
  "Nuez moscada",
  "Sal",
  "Pimienta",
]
steps = [
  "Preparar las placas de lasaña y reservarlas sobre un paño limpio para que no se peguen.",
  "Pelar y cortar en daditos la cebolla y la parte blanca del puerro.",
  "Dorarlas en una cazuela grande con un poco de aceite.",
  "Añadir la calabaza cortada en dados, sin piel ni pepitas. salpimentar y rehogar unos minutos.",
  "Poner una tapa y dejar el fuego medio-bajo hasta que la calabaza esté tierna.",
  "Añadir las avellanas rotas en trozos grandes y las pasas.",
  "Calentar la leche con la rama de tomillo entera.",
  "En una cazuela mediana, fundir la mantequilla y un poco de aceite a fuego suave unos 10´, hasta que esté traslúcida.",
  "Añadir la harina, remover y dejar que se haga a fuego muy lento otros 10´.",
  "Incorporar la leche bien caliente removiendo con una cuchara de madera hasta obtener una bechamel ligera.",
  "Añadir el gorgonzola y removerlo hasta que se funda y mezcle bien.",
  "Salpimentar, sazonar con nuez moscada rallada al gusto y retirar el tomillo.",
  "En un molde apto para horno  de unos 30x20 cm poner una capa fina de bechamel, una capa de pasta, un tercio del relleno de calabaza y un poco de bechamel.",
  "Hacer 3 capas de relleno, terminando con placas y más bechamel encima.",
  "Cubrir con el parmesano y llevar al horno a 180 grados unos 20´o  hasta que el gratinado esté dorado.",
  "Servir inmediatamente.",
]
