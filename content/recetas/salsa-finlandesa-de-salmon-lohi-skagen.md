categories = [
  "Salsas",
]
author = "Amama"
difficulty = "Baja"
time = "Menos de 30 min"
ingredients = [
  "150gr de salmón ahumado cortado en trozos pequeños",
  "240gr de crème fraîche (crema agria)",
  "2 a 4 cucharadas de mayonesa",
  "1 cebolla roja muy picada",
  "Eneldo fresco y cebollino picados",
  "1 cucharadita de mostaza (opcional)",
  "Un chorrito de zumo de limón",
  "Sal y pimienta",
  "Pan de centeno o pan blanco tostado",
  "Huevas de trucha o salmón para decorar (opcional)",
]
steps = [
  "Batir ligeramente la crema fresca con la batidora durante 30' hasta que cojaun poco de consistencia.",
  "Añadir la mayonesa, la cebolla roja picada, el eneldo, el cebollino, la mostaza y el limón",
  "Agregar el salmón troceado con cuidado para que no se rompa demasiado el pescado.",
  "Salpimentar al gusto.",
  "Tostar el pan y colocar una porción generosa por encima.",
  "Decorar con un poco más de eneldo y huevas por encima.",
]
