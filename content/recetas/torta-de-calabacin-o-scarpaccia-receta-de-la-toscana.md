title = "Torta de calabacín o Scarpaccia (receta de la Toscana)"
date = "2026-08-19T18:03:15"
categories = [
  "Verduras",
]
author = "Amama"
difficulty = "Baja"
time = "60-120 min"
servings = "4-6 personas"
ingredients = [
  "3 calabacines medianos ( unos 750gr)",
  "1 cebolla",
  "120gr de harina integral",
  "Aceite de oliva",
  "Pimienta negra",
  "Sal",
  "50gr de queso parmesano ( opcional)",
]
steps = [
  "Cortar los calabacines en rodajas finas.",
  "Picar la cebolla en juliana fina.",
  "Por todo en un escurridor o colador sobre un plato ondo o un bol.",
  "Salar y remover bien, dejar reposar durante 2 horas como mínimo, con peso encima.",
  "Calentar el horno a 220º",
  "Poner un bol grande con la harina y añadir el líquido que hayan soltado los calabacines, una cucharada de aceite, pimienta negra y el queso si se quiere.",
  "Mezclar todo.",
  "Añadir los calabacines y la cebolla y mezclar.",
  "Verter la mezcla en una bandeja de horno de unos 35×35 cm forrada con papel de horno y untada con aceite.",
  "Conviene aplastar la masa ligeramente con las manos para repartirla y compactar, tiene que quedar una capa de unos 2cm aproximadamente.",
  "Añadir un chorro de aceite de oliva por encima y hornear durante 40' o hasta que la parte superior esté bien dorada.",
  "Dejar que se temple fuera del horno sobre una rejilla.",
  "Opcionalmente se puede dar la vuelta y dorar por el otro lado 5' en la parte alta del horno y al grill.",
  "Servir templada o a temperatura ambiente.",
]
+++

Si no se tiene bandeja de ese tamaño, se puede hacer en 2.
