title = "Empanadas de queso y cebollas"
date = "2024-06-08T16:27:50"
categories = [
  "Aperitivos y tapas",
]
author = "Amama"
difficulty = "Baja"
time = "30-60 min"
ingredients = [
  "Para la masa:",
  "500 gr de harina de trigo.",
  "125 gr de mantequilla.",
  "170 ml de agua.",
  "1 cucharadita de sal.",
  "Para el relleno:",
  "500 gr de cebollas.",
  "300 gr de mozzarella o queso Arzúa-Ulloa.",
  "150 gr de algún queso más potente rallado (grana parmesano, gorda,",
  "1 cucharadita de tomillo seco ( u orégano).",
  "Sal.",
  "Opcional: yema de huevo batida.",
]
steps = [
  "Pelar y cortar la cebolla en juliana y rehogarla lentamente con un poco de aceite.",
  "Añadir la sal y el tomillo. Retirar cuando esté blanda y traslúcida. Escurrir y enfriar.",
  "Para preparar la masa: mezclar la harina, la sal 100 gr de mantequilla y el agua.",
  "Amasar hasta conseguir una bola de textura lisa.",
  "Estirar hasta conseguir un cm de espesor y untar con el resto de la mantequilla.",
  "Hacer 3 pliegues y estirar hasta obtener una lámina de unos 5 mm de espesor.",
  "Cortar unos discos de unos 10-12 cm de diámetro.",
  "Calentar el horno a 180º.",
  "Preparar el relleno mezclando la mozzarella o queso Arzúa-Ulloa cortado en daditos, el queso rallado y la cebolla fría.",
  "Rellenar las empanadas con con una cucharada de la mezcla.",
  "Poner agua en uno de los bordes para que peguen al cerrarlas y el repulgue.",
  "Colocar las empanadas en una bandeja de horno con papel vegetal o aceite ( se pueden pincelar con yema de huevo batido).",
  "Hornear durante 20’ o hasta que estén doradas.",
  "Servir calientes o tibias.",
]
