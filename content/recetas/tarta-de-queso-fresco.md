date = "2024-05-20T14:23:25"
categories = [
  "Postres",
]
author = "Amama"
difficulty = "Baja"
time = "30-60 min"
ingredients = '''
1 kg de queso fresco.
20 cucharadas de azúcar.
2 cucharadas de maizena.
6 huevos.
1 vaso de leche.
100 gr de pasas de Corinto.'''
preparation = '''
Macerar las pasas en algún licor ( coñac).
Mezclar el resto de ingredientes menos la leche y batir.
Añadir la leche y mezclar.
Poner el contenido en un molde de horno enharinado y hornear a 180º durante 15’
Sacar del horno y añadir las pasas.
Volver a introducir en el horno hasta que cuaje.'''
