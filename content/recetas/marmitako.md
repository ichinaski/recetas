ingredients = [
  "500 gr de bonito",
  "500 gr de patatas",
  "1 zanahoria mediana",
  "2 pimientos verdes medianos",
  "2 pimientos choriceros",
  "1,5 cebollas",
  "2 dientes de ajo",
  "1 tomate maduro",
  "100 ml de vino blanco",
  "1 cucharadita de pimentón dulce",
  "Aceite de oliva",
  "Sal",
]
steps = [
  "Rehidratar los pimientos choriceros poniéndolos en un bol y echándoles agua hirviendo hasta cubrirlos.",
  "Picar fino la cebolla, el pimiento verde y la zanahoria.",
  "Pelar despepitar y picar el tomate.",
  "Rehogar la cebolla, el pimiento verde y la zanahoria en una cazuela con un chorro de aceite de oliva y una pizca de sal, unos 20’ a fuego medio- suave.",
  "Añadir el ajo picado y rehogar 5’ más.",
  "Añadir el tomate y dejar 5’ más.",
  "Romper los pimientos choriceros en 2 partes verticalmente. Despegar la carne de la piel con una cuchara y desechar la sal pepitas y la piel.",
  "Pelar las patatas, cascarlas en trozos como de bocado.",
  "Añadir el pimentón y el pimiento choricero, remover y añadir las patatas.",
  "Rehogarlas unos 10’ removiendo de vez en cuando.",
  "Mojar con el vino y dejar que evapore el alcohol 1’",
  "Cubrir con agua, salar y dejar a fuego suave durante una hora aproximadamente, moviendo de vez en cuando.",
  "El caldo debe quedar gordito, pero no demasiado espeso. Si está muy líquido, aplastar un par de trozos de patata y disolverlos, y si está muy pastoso, añadir agua.",
  "Retirar del fuego y añadir el bonito cortado en trozos de 2cm.",
  "Tapar y dejar que se haga con el calor durante unos 5’ más.",
  "Corregir de sal y servir caliente o templado.",
]
