title = "Tartaleta de queso y manzana"
date = "2026-01-30T10:18:37"
categories = [
  "Postres",
]
author = "Amama"
difficulty = "Baja"
time = "30-60 min"
ingredients = [
  "1 paquete de masa para empanadillas",
  "1 tarrina de queso tipo Philadelphia",
  "3 huevos",
  "3-4 manzanas",
  "mermelada de melocotón",
  "50-100gr de azúcar",
  "30gr de mantequilla",
  "1 cucharadita de vainilla (opcional)",
]
steps = [
  "Precalentar el horno a 200º (calor arriba y abajo).",
  "Pelar las manzanas y cortarlas en trocitos.",
  "Hacer la crema de queso:  poner en una cazuela el queso, los huevos, el azúcar, la mantequilla y la vainilla.",
  "Triturar con la batidora y poner a fuego medio-bajo, sin parar de remover hasta que espese (tiene que quedar como unas natillas no muy espesas).",
  "Poner las masas colocadas en moldes para magdalenas, adaptándolas a su forma.",
  "Poner en el fondo 2 trocitos de manzana, cubrir con la crema de queso y repartir una o dos láminas de manzana sobre la superficie.",
  "Hornear 25'-30'.",
  "Dejar enfriar, desmoldar y untar con un poco de mermelada de melocotón diluida con un poco de agua.",
]
