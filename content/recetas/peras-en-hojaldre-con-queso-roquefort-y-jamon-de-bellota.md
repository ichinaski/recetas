  "Aperitivos y tapas",
]
author = "Amama"
difficulty = "Baja"
time = "30-60 min"
servings = "4 personas"
ingredients = [
  "4 peras (mejor verdes)",
  "1 plancha de hojaldre",
  "200 gr de queso (se puede utilizar otro tipo de queso como azul, camember, brie.....)",
  "8 lonchas de jamón ibérico u otro que se quiera",
  "8 nueces peladas",
  "1 huevo batido para pintar el hojaldre",
]
steps = [
  "Precalentar el horno arriba y abajo a 180º",
  "Lavar la peras, cortar el rabito y partir por la mitad a lo largo.",
  "Hacer un hueco en el centro de a pera, donde están las pepitas, para luego rellenar con queso.",
  "Colocar sobre la bandeja un papel de horno y poner las peras con el agujero hacia abajo.",
  "Pinchar el hojaldre con un tenedor para evitar que se hinche.",
  "Cortar la plancha de hojaldre en rectángulos (usar la medida de una pera) y tapar la pera con el hojaldre.",
  "Batir el huevo y pintar todo el hojaldre.",
  "Hornear 25’ y sacar del horno.",
  "Darles la vuelta sobre la misma bandeja, rellenar el hueco con trozos de queso y colocar una nuez encima.",
  "Meter de nuevo en el horno y dejar 3’-4’ y sacar del horno.",
  "Dejar atemperar un poco y colocar en cada pera un trozo de jamón.",
]
