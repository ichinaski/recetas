title = "Pantxineta, tarta de hojaldre tradicional vasca"
date = "2024-07-20T19:46:02"
categories = [
  "Postres",
]
author = "Amama"
difficulty = "Baja"
time = "30-60 min"
servings = "6 personas"
ingredients = [
  "2 masas de hojaldre",
  "50 gr de almendras laminadas",
  "Azúcar glass para decorar",
  "Frambuesa, mora y menta ( para decorar)",
  "Crema pastelera:",
  "350 ml de leche",
  "150 ml de nata",
  "1 rama de canela",
  "1 limón",
  "40 gr de maicena (harina maíz refinada)",
  "75 gr de azúcar",
  "1 huevo",
  "2 yemas de huevo",
]
steps = [
  "Crema pastelera:",
  "En una cazuela poner a calentar la leche, la nata, la rama de canela troceada y al corteza de limón. Dejar hervir a fuego medio.",
  "En un bol poner el azúcar y la maicena y mezclar con las varillas.",
  "Agregar las 2 yemas de huevo y el huevo. Remover hasta que quede una mezcla homogénea.",
  "Colocar un colador sobre el bol y verter la leche infusionada, retirando la canela y al piel de limón. Remover todo con las varillas.",
  "Verter la mezcla de nuevo a la cazuela.",
  "Poner a fuego suave y seguir removiendo con las varillas enérgicamente hasta que hierva y se espese la crema.",
  "Cuando hierva, retirar del fuego y verter la crema pastelera en un bol.",
  "Tapar el bol con film, tocando la crema, meter en el frigorífico y dejar que repose antes de su uso.",
  "Extender una masa de hojaldre y colocarla en una bandeja con papel de horno.",
  "Colocar encima la crema pastelera y repartirla de manera uniforme, dejando libres los bordes.",
  "Con ayuda de un pincel, mojar ligeramente los bordes de la masa con agua.",
  "Tapar con la otra masa de hojaldre y presionar en los bordes ligeramente para unir las 2 masas.",
  "Mojar toda la superficie con agua utilizando el pincel.",
  "Repartir las almendras laminadas por toda la superficie.",
  "Precalentar el horno a 180º e introducir la pantxineta unos 25’ aproximadamente, hasta que empiece a dorarse.",
  "Sacar la tarta del horno y dejar templar.",
  "Espolvorear la superficie de la tarta con azúcar glass.",
  "Servir la pantxineta y decorarla con una Mora, una frambuesa y una hoja de menta.",
]
+++

Se pueden hacer otras opciones para rellenar: trufa, nata montada de chocolate, crema de chocolate, crema pastelera de limón, crema de almendras.....
