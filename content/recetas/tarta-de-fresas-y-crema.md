title = "Tarta de fresas y crema"
date = "2026-08-21T18:43:37"
categories = [
  "Postres",
]
author = "Amama"
difficulty = "Baja"
time = "30-60 min"
ingredients = [
  "Una plancha rectangular de hojaldre de mantequilla",
  "500gr de fresas",
  "70ml de leche entera",
  "70gr de harina",
  "100gr de azúcar",
  "2 huevos",
  "3 yemas",
  "1 vaina de vainilla",
  "1 trozo de piel de limón ( opcional)",
]
steps = [
  "Calentar el horno a 200º.",
  "Colocar la plancha en el molde con papel de horno, dejando los bordes con 3cm de alto aproximadamente.",
  "Pinchar con tenedor varias veces.",
  "Colocar encima otro papel de horno y peso ( unos garbanzos secos pueden servir)",
  "Hornear 30'-35'.",
  "Retirar los pesos y hornear 5' más si la superficie está muy blanca. Dejar enfriar.",
  "Preparar la crema pastelera:",
  "Calentar la leche con la vainilla y la piel de limón en un cazo a fuego medio-bajo.",
  "Batir los huevos, las yemas, la harina y el azúcar hasta que esté todo bien incorporado.",
  "Cuando la leche esté caliente, añadir al bol, mezclar bien y devolver la mezcla al cazo.",
  "Cocinar sin dejar de remover hasta que espese durante 5' aproximadamente.",
  "Retirar el limón y la vainilla,  tapar con film en contacto con la crema para que no se seque la superficie y dejar templar.",
  "Limpiar y cortar las fresas por la mitad, o en cuartos o láminas.",
  "Verter la crema pastelera sobre el hojaldre y colocarvlas fresas por encima.",
  "Conservar en la nevera.",
]
