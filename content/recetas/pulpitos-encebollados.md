categories = [
  "Pescado",
]
author = "Amama"
difficulty = "Baja"
time = "30-60 min"
servings = "5-6 personas"
ingredients = [
  "1,2 kg de pulpitos limpios (pueden ser descongelados)",
  "3 cebollas grandes o 4 medianas",
  "4 dientes de ajo",
  "400ml de vino blanco",
  "Aceite de oliva",
  "Sal",
  "Copos de chile o pimentón",
  "Perejil (opcional)",
  "Arroz, patatas, polente, cuscus o pan para acompañar",
]
steps = [
  "Limpiar los pulpitos, si son frescos y escurrirlos bien.",
  "Pelar las cebollas y cortarlas en tiras.",
  "Pelar los ajos y laminarlos.",
  "Pochar todo en una cazuela con un poco de aceite y sal a fuego medio/bajo.",
  "Añadir los pulpitos, subir un  poco el fuego y dar vueltas durante 2'-3'.",
  "Añadir el vino blanco y el chile o pimentón al gusto y llevar a ebullición,  cuando hierva, bajar el fuego al mínimo y dejar cocinar durante 45'.",
  "Se puede ir tapando y destapando la cazuela para conseguir una salsa reducida, pero que no se seque (si se seca demasiado,  añadir un poco más de vino o agua).",
  "Pasado el tiempo, probar si los pulpitos ya están tiernos, si no , dejarlos 10' más y volver a probar.",
  "Servir con el acompañamiento deseado.",
]
+++

Se puede congelar si sobra.
Se le pueden añadir garbanzos, alubias o con pasta larga, tipo espaguetis o tallarines.
Va bien también con un poco de ralladura de limón.
