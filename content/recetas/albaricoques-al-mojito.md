  "Postres",

author = "Amama"
difficulty = "Baja"
time = "30-60 min"
servings = "2 personas"
ingredients = [
  "6 albaricoques.",
  "El zumo de una lima.",
  "Una sartén hojas de hierbabuena.",
  "1 chupito de ron blanco.",
  "1 cucharada de azúcar.",
]
steps = [
  "En un mortero, moler la hierbabuena ( reservando algunas hojas para emplazar con el azúcar ).",
  "Añadir el zumo de lima, el ron y remover.",
  "Cortar los albaricoques por la mitad y deshuesarlos.",
  "Ponerlos en un recipiente con la mezcla del mortero, pasada por un colado para retirar los restos de hierbabuena.",
  "Añadir 2 cucharadas de agua, remover para que se impregnen los albaricoques.",
  "Tapar con film y macerar en la nevera 2 horas.",
  "Colocar los albaricoques en una fuente de horno y cubrir con el líquido sobrante.",
  "Hornear a 180º durante 10’. Añadir una pizca de azúcar por encima y hornear con el grill hasta que se doren ligeramente.",
  "Servir con unas hojas de hierbabuena por encima. Se puede acompañar de algún queso tipo mató o requesón.",
]
