title = "Crema de patata, pulpo y huevo"
date = "2026-01-25T15:33:22"
categories = [
  "Pescado",
]
author = "Amama"
difficulty = "Baja"
time = "30-60 min"
servings = "4 personas"
ingredients = [
  "500gr de patada sin piel y cortadas en dados medianos",
  "200ml de nata líquida o leche evaporada",
  "150ml de agua de hervir las patatas",
  "60ml de aceite de oliva virgen extra",
  "250gr de pulpo cocido cortado en rodajas",
  "4 huevos",
  "Sal",
  "Pimienta",
  "Pimentón de la vera (dulce o picante)",
]
steps = [
  "Hervir las patatas en agua ligeramente salada durante 10' o hasta que estén tiernas.",
  "Escurrir y reservar 100ml del agua de cocción.",
  "Mezclar el agua de cocción con la nata ( o leche evaporada), el aceite de oliva, sal y pimienta.",
  "Procesar las patatas con un pasapurés o, presionando.",
  "Mezclar las patatas, la nata o leche, el agua y el aceite con unas varillas (no usar batidora porque se liberaría demasiado almidón ).",
  "Rectificar de sal y pimienta.",
  "Hacer los huevos escalfados o mollet.",
  "Servir el huevo sobre el puré  caliente, acompañado del pulpo también caliente, con un punto de sal y pimentón de la Vera.",
  "Comer inmediatamente.",
]
