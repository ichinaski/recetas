date = "2025-10-29T19:00:28"
categories = [
  "Verduras",
]
author = "Amama"
difficulty = "Baja"
time = "30-60 min"
ingredients = [
  "1 kg de calabaza",
  "4 puerros",
  "Aceite de oliva Virgen extra",
  "20gr de mantequilla",
  "50gr de avellanas peladas y tostadas",
  "Tomillo fresco u otras hierbas",
  "Sal fina",
  "Sal en escamas",
  "Pimienta negra",
]
steps = [
  "Precalentar el horno a 180º",
  "Limpiar y eliminar la parte verde de los puerros (guardar para otras preparaciones)",
  "Cortar la parte blanca en trozos de 6-7 cm.",
  "Colocar los puerros en un cazo y cubrir con aceite de oliva y añadir una ramita de tomillo fresco.",
  "Cocinar a fuego muy suave durante 50' o hasta que estén tiernos.",
  "Lavar y cortar la calabaza en trozos grandes y retirar las semillas.",
  "Colocar en una bandeja de horno con la piel hacia abajo.",
  "Añadir aceite de oliva, sal, pimienta y remover con las manos para que quede bien repartido.",
  "Hornear 40' o hasta que esté blanda.",
  "Pelar la calabaza cuando se haya enfriado un poco y colocarla en un bol.",
  "Aplastar con un tenedor,  añadir la mantequilla, sal y pimienta.",
  "Colocar el puré en una bandeja como base.",
  "Retirar los puerros del aceite, bien escurridos y colocarlos encima del puré.",
  "Terminar con tomillo fresco, las avellanas picadas, sal en escamas y pimienta negra recien molida.",
]
