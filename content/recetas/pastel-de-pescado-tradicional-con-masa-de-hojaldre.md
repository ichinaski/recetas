title = "Pastel de pescado tradicional con masa de hojaldre"
date = "2025-05-03T16:53:40"
categories = [
  "Pescado",
]
author = "Amama"
difficulty = "Baja"
time = "30-60 min"
ingredients = [
  "400gr de bacalao sin espinas",
  "1 cebolla",
  "1 hoja de laurel",
  "400 ml de leche 25gr de mantequilla",
  "25gr de harina",
  "Una cucharada de perejil seco",
  "Un cucharada de mostaza en polvo",
  "Una lata de atún de 200gr",
  "10 gambones cocidos",
  "250gr de masa de hojaldre redonda",
  "1 huevo batido para glasear",
  "Sal y pimienta",
]
steps = [
  "Pelar y cortar la cebolla en cuartos.",
  "Cortar en cubos pequeños la mantequilla.",
  "Precalentar el horno a 180°",
  "Colocar el bacalao, la cebolla y la hoja de laurel en una sartén y echar la leche encima.",
  "Cocinar a fuego medio 8'",
  "Retirar el pescado, la cebolla y la hoja de laurel.",
  "Añadir la mantequilla a la leche y remover a fuego lento hasta que se derrita.",
  "Incorporar la harina, el perejil y la mostaza,  batiendo hasta obtener una salsa espesa y brillante evitando grumos.",
  "Salpimentar.",
  "Añadir el bacalao cocido,  los gambones y el atún a la salsa.",
  "Poner los ingredientes en una fuente o molde de horno.",
  "Colocar la masa de hojaldre sobre la mezcla de pescado, hacer un pequeño agujero en el centro para que salga el vapor y no se agujeree la masa.",
  "Pintar el hojaldre con huevo batido.",
  "Hornear 30' hasta que esté dorado.",
]
+++

Se puede sustituir el bacalao por cualquier otro pescado blanco.
