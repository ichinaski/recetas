ingredients = [
  "1 kg de patata",
  "4 zanahorias",
  "4 huevos duros",
  "1 pata de pulpo cocido (250gr)",
  "100gr de aceitunas rellenas de anchoa",
  "16 piparras",
  "300gr de mayonesa (o al gusto)",
  "Un chorro de vinagre de las piparras (opcional)",
  "Para la salsa:",
  "12 piparras 4 cucharadas de aceite de oliva",
  "Un chorro de vinagre de las piparras (al gusto)",
  "Además,  para servir:",
  "Picos, regañas o mariñeiras.",
]
steps = [
  "Preparar la salsa de piparras,  quitando el pedúnculo  y triturando con el resto de ingredientes hasta conseguir una salsa fina.",
  "Cocer las patatas bien limpias con piel en agua, unos 20', dependiendo del tamaño.",
  "Cocinar también las zanahorias enteras unos 15', dependiendo del tamaño, pero que no queden muy blandas.",
  "Pelar y trocear las patatas y poner en una ensaladera.",
  "Trocear las zanahorias y poner en la ensaladera.",
  "Cocer 4 huevos, pelar y reservar.",
  "Trocear al pata de pulpo en trozos como de 0,5cm y añadir a las patatas tibias.",
  "Cortar 8  piparras en rodajas, las aceitunas y el huevo duro y unir a la ensaladera.",
  "Preparar la mayonesa usando el vinagre de las piparras.",
  "Mezclar bien la ensaladilla con la mayonessa, rectificar de aderezo (sal, acidez..) y llevar a la nevera al menos 2 horas.",
  "Servir con las piparras reservadas, la salsa al gusto y mariñeiras u otro pan crujiente.",
