categories = [
  "Sopas y cremas",
]
author = "Amama"
difficulty = "Baja"
time = "30-60 min"
servings = "4 personas"
ingredients = [
  "4 pepinos medianos (unos 600 gr)",
  "1/2 melón piel de sapo (unos 700 gr)",
  "200 gr de leche de coco",
  "200 ml de agua",
  "Una cucharada rasa o al gusto de curry en polvo",
  "Sal",
  "Zumo de limón al gusto",
  "1 diente pequeño de ajo",
  "50 gr de pistachos pelados",
]
steps = [
  "Quitar las puntas y la mitad de la piel a tiras a los pepinos y cortarlos en dados.",
  "Pelar el diente de ajo.",
  "Pelar el melón, quitarle las semillas y cortarlo en dados.",
  "Poner todo en un vaso batidor, menos los pistachos.",
  "Triturar hasta conseguir una textura emulsionada, probar y rectificar de sal.",
  "Guardar en la nevera al menos 2 horas o dejar para día siguiente.",
  "Servir con los pistachos picados.",
]
