difficulty = "Baja"
time = "30-60 min"
servings = "6 raciones"
ingredients = [
  "400 ml de nata para montar (35%)",
  "80 gr de azúcar glass",
  "12 bizcochos de soletilla (2 por vaso)",
  "Caramelo líquido",
  "Para la crema:",
  "1/2 litro de leche",
  "60 gr de azúcar",
  "4 yemas de huevo",
  "20 gr de  maizena",
  "1 taponcito de extracto de vainilla",
  "Cáscara de limón",
  "1 rama de canela",
  "Para el almíbar:",
  "150  ml de agua",
  "150 gr de azúcar",
  "1 poquito de ron tostado",
  "1 cc de esencia de vainilla",
]
steps = [
  "Poner en una cazuela a fuego medio la leche, vainilla, piel de limón y rama de canela.",
  "Cuando empiece a hervir, retirar del fuego, tapar e infusionar 15’",
  "En un recipiente batir las yemas con el azúcar hasta que la mezcla blanquee un poco.",
  "Poner la maizena con un poco de leche, diluir y verter en la mezcla de yemas.",
  "Colar la leche infusionada y verter sobre la mezcla preparada.",
  "Volver a poner a fuego medio y batir con varillas hasta que comience a espesar. Dejar atemperar y guardar en el frigorífico cubierto con papel film unas horas.",
  "Para el almíbar:",
  "Poner el agua y azúcar en un cazo al fuego y dejar hervir unos 8’ hasta que espese ligeramente, añadir el ron tostado y la esencia de vainilla.",
  "Para el montaje:",
  "Colocar un bizcocho embebido en el fondo, agregar nata montada, cubrir nuevamente con otro bizcocho y rellenar con la crema por encima ( batir antes de incorporar).",
  "Terminar con un poco de caramelo líquido.",
]
