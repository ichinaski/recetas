categories = [
  "Pasta",
]
author = "Amama"
difficulty = "Baja"
time = "30-60 min"
servings = "4 personas"
ingredients = [
  "500gr de espaguetis",
  "1 calabacín",
  "8 espárragos trigueros",
  "150gr de guisantes frescos o congelados",
  "150gr de setas al gusto",
  "2 pimientos rojos italianos",
  "3 dientes de ajo",
  "600gr de tomate troceado",
  "Aceite de oliva virgen",
  "120ml de leche evaporada (o nata líquida )",
  "150gr de queso parmesano rallado",
  "Sal y pimienta al gusto",
  "orégano o albahaca fresco",
]
steps = [
  "Limpiar y cortar el calabacín en dados con piel. Pasarlo por el microondas tapado, unos minutos (8) para quitarle el agua y reservar.",
  "Limpiar los trigueros, retirar el tallo y trocear. Pasarlos por el microondas 5' si son delgados, o algo más si son gruesos. Reservar.",
  "Limpiar las setas, secar y trocear.",
  "Cocer unos minutos los guisantes. Cortar la cocción con agua fria y reservar.",
  "Limpiar los pimientos y cortar en tiritas.",
  "Pelar y majar los dientes de ajo.",
  "Cubrir el fondo de una cazuela con un poco de aceite y sofreir el pimiento hasta que cambie de color.",
  "Agregar el calabacín,  los trigueros y las setas. Cocinar unos minutos, hasta que el calabacín acabe de hacerse y las setas se cocinen.",
  "Agregar el ajo majado, darle unas vueltas y añadir el tomate troceado y los guisantes. Reducir unos minutos el líquido del tomate.",
  "Una vez espesado el sofrito, dejarlo tapado y cocer la pasta en abundante agua salada hasta que esté al dente.",
  "Escurrir la pasta y añadirla inmediatamente al sofrito.",
  "A fuego medio agregar la leche evaporada o nata y el queso rallado. Remover todo y sazonar al gusto con sal y pimienta.",
  "Servir enseguida poniendo por encima orégano o albahaca frescos.",
]
+++

Admite cualquier tipo de verdura, judias verdes, brecol, zanahoria......o cualquier verdura de temporada.
Se puede utilizar cualquier tipo de pasta.
