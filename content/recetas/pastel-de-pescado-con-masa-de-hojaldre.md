author = "Amama"
difficulty = "Media"
time = "30-60 min"
ingredients = [
  "400gr de bacalao sin espinas",
  "1 cebolla",
  "1 hoja de laurel",
  "400ml de leche",
  "25 gr de mantequilla",
  "25 gr de harina de trigo",
  "1 cucharada de perejil seco",
  "1 cucharada de mostaza",
  "1 lata de atún (200gr)",
  "10 gambones cocidos",
  "250gr de masa de hojaldre redonda",
  "1 huevo batido para glasear",
  "Sal y pimienta",
]
steps = [
  "Pelar y cortar la cebolla en cuartos.",
  "Cortar en cubos pequeños la mantequilla.",
  "Precalentar el horno a 180°",
  "Colocar el bacalao, la cebolla y la hoja de laurel en una sartén y verter la leche encima.",
  "Cocinar a fuego lento durante 8'",
  "Retirar el pescado, la cebolla y la hoja de laurel dela sartén.",
  "Añadir la mantequilla a la leche y remover a fuego lento hasta que se derrita.",
  "Incorporar la harina, el perejil seco y la mostaza batiendo hasta  obtener una salsa espesa y brillante sin que tenga grumos.",
  "Salpimentar.",
  "Agregar el bacalao cocido, los gambones y el atún a la salsa.",
  "Colocar la masa de hojaldre sobre la mezcla de pescado, hacer un agujero en el centro para que salga el vapor y no se agujeree la masa en el horno ( se puede reservar algo de masa para decorar.",
  "Pintar el hojaldre con huevo batido y hornear durante 30', hasta que esté dorado.",
]
+++

Se puede sustituir el bacalao por cualquier otro pescado blanco.
