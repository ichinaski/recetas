  "2 cucharadas de mantequilla",
  "1 puerro",
  "1 o 2 zanahorias",
  "750ml de caldo de pescado",
  "200ml de nata para cocinar (o leche evaporada )",
  "500gr de salmón",
  "2 patatas",
  "Eneldo fresco para servir",
  "Sal",
  "Pimienta negra",
]
steps = [
  "Picar el puerro y la zanahoria.  Añadir el puerro junto con la mantequilla en una olla y pochar a fuego medio-bajo.",
  "Cubrir con el caldo.",
  "Añadir las patatas y la zanahoria peladas y cortadas em dados.",
  "Salpimentar.",
  "Dejar cocer a fuego medio durante 10'-15' hasta que las patatas estén tiernas.",
  "Añadir la nata (o leche evaporada) y el salmón cortado en trozos de bocado.",
  "Cocinar 2'.",
  "Servir con eneldo y pimienta negra.",
